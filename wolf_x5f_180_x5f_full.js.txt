const rl = readline.createInterface({input:process.stdin,output:process.stdout})
const q=t=>new Promise(r=>rl.question(t,r))

async function handleAghnya(sock, from, query, msg){
 if(!query){ await sock.sendMessage(from,{text: box10("🎧","اغنية",".اغنية","يبحث ويبعت الأغنية MP3",".اغنية اسم","الجودة: 320kbps",".اغنية تملي معاك","اكتب اسم الاغنية")}); return}
 await sock.sendMessage(from,{text:`🎧 جاري البحث عن: *${query}* ...`})
 try{
  const search = await yts(query)
  const video = search.videos[0]
  if(!video){ await sock.sendMessage(from,{text:'❌ مش لاقي الاغنية'}); return}
  await sock.sendMessage(from,{text:`🎧 لقيتها: ${video.title}
⏳ جاري التحميل...`})
  // هنا تبعت لينك يوتيوب والمفروض بلجن ytdl يشتغل
  await sock.sendMessage(from,{text:`${video.url}`})
 }catch(e){ await sock.sendMessage(from,{text:'❌ خطأ في البحث'})}
}

async function handleSora(sock, from, query){

 await sock.sendMessage(from,{text:`🕋 جاري جلب سورة *${query}* ...`})
}

function fullMenuText(){

  txt+=`${d.e} ${i}. ${k} - ${d.t}\n`
  i++

}

async function startBot(){
 const {version}=await fetchLatestBaileysVersion()
 const {state,saveCreds}=await useMultiFileAuthState(sessionPath)
 const sock=makeWASocket({version,auth:state,logger:P({level:'silent'}),browser:["Wolf","Chrome","1.0"]})
 sock.ev.on('creds.update',saveCreds)
 if(!state.creds.registered){
  const phone=await q('\n🐺 رقمك مع كود البلد: ')
  const code=await sock.requestPairingCode(phone.trim())
  console.log('\n🔥 كود الاقتران: '+code+'\n')
 }
 sock.ev.on('connection.update',u=>{
  if(u.connection==='open'){console.log('✅ ONLINE Hamo & Karim - 180 امر - 10 سطور');rl.close()}
  if(u.connection==='close'&&u.lastDisconnect?.error?.output?.statusCode!==DisconnectReason.loggedOut)startBot()
 })
 sock.ev.on('messages.upsert',async({messages})=>{
  const msg=messages[0]; if(!msg.message) return
  const from=msg.key.remoteJid
  let text=(msg.message.conversation||msg.message.extendedTextMessage?.text||'').trim()
  if(!text) return
  // بوت اوامر فقط: لازم يبدأ بنقطة
  if(!text.startsWith('.')) return
  text=text.replace(/\.+$/,'').trim()
  let [cmd,...args] = text.split(' ')
  cmd=cmd.toLowerCase()
  const query = args.join(' ')
  if(!ALL[cmd]) return // مش امر من الـ 180 -> ميردش
  const d=ALL[cmd]
  try{ await sock.sendMessage(from,{react:{text:d.r,key:msg.key}})}catch{}
  if(cmd==='.منيو'){ await sock.sendMessage(from,{text:fullMenuText()}); return}
  if(cmd==='.اغنية'){ await handleAghnya(sock,from,query,msg); return}
  if(cmd==='.سورة'){ await handleSora(sock,from,query); return}
  // باقي الاوامر 10 سطور مزخرفة
  const box = box10(d.e,d.t,d.c,d.f,d.u,d.d+': '+d.q,d.ex,d.n)
  await sock.sendMessage(from,{text:box})
 })
}
startBot()
